type Sample = { model?: string; totalTokens?: number; inputTokens?: number; cacheReadInputTokens?: number; cacheCreationInputTokens?: number; outputTokens?: number; contextWindow?: number };
const update = (context: unknown, chatId = "context-frozen") => ({ chatId, type: "contextUsage", context }) as unknown as ReasonPayload;
const settle = () => new Promise<void>((r) => setImmediate(r));

function runtime(shared = new Map<string, string>()) {
  let deliver: Deliver = () => assert.fail("listen first");
  const backend: ChatBackend<null> = {
    canSteer: false, defaultConfig: null,
    sessionKey: (id) => `session.${id}`, configKey: (id) => `config.${id}`, parseConfig: () => null,
    async listen(fn) { deliver = fn; }, async open() {}, async prompt() { return {}; }, cancel() {},
  };
  const host: ChatHost = {
    persist: false, async invoke() { return null; }, recordTurnChanges() {},
    storage: { getItem: (k) => shared.get(k) ?? null, setItem: (k, v) => { shared.set(k, v); }, removeItem: (k) => { shared.delete(k); } },
  };
  const rt = createChatRuntime(backend, host);
  rt.subscribe("context-frozen", () => {});
  return { rt, shared, async emit(context: unknown, id = "context-frozen") {
    await settle();
    const payload = update(context, id);
    const step = applyReasonPayload(reasonChat(id), payload);
    assert.ok(step, "actual payload reducer must accept the additive context sample without a transcript event");
    deliver(id, (state) => applyReasonPayload(state, payload));
  }, context(id = "context-frozen"): Sample | undefined {
    return (rt.snapshot(id) as unknown as { contextUsage?: Sample }).contextUsage;
  } };
}

test("context telemetry is a distinct pure step, not a cumulative usage replacement", () => {
  const context = { kind: "sample", model: "claude-test", inputTokens: 20, cacheReadInputTokens: 50, cacheCreationInputTokens: 10, outputTokens: 5 };
  const state = reasonChat("context-frozen");
  const result = applyReasonPayload(state, update(context)) as Step & { context?: unknown };
  assert.ok(result, "context payload must return a real Step");
  assert.deepEqual(result.context, context);
  assert.deepEqual(result.events, [], "context must not manufacture a message or alter turn cost");
  assert.equal(result.state, state);
});

test("actual runtime keeps latest cache-inclusive sample and same-model denominator across reload", async () => {
  const f = runtime();
  await f.emit({ kind: "window", model: "claude-test", contextWindow: 100 });
  await f.emit({ kind: "sample", model: "claude-test", inputTokens: 20, cacheReadInputTokens: 50, cacheCreationInputTokens: 10, outputTokens: 5 });
  assert.equal(f.context()?.totalTokens, 85);
  assert.equal(f.context()?.contextWindow, 100);
  await f.emit({ kind: "sample", model: "claude-test", inputTokens: 10, cacheReadInputTokens: 20, outputTokens: 1 });
  assert.equal(f.context()?.totalTokens, 31, "latest top-level sample replaces, not adds to prior context");
  assert.equal(f.rt.snapshot("context-frozen").transcript.messages.length, 0);
  const reloaded = runtime(f.shared);
  assert.equal(reloaded.context()?.totalTokens, 31, "the context-only update must survive runtime recreation");
  assert.equal(reloaded.context()?.contextWindow, 100);
  assert.equal(reloaded.context("another-chat"), undefined, "context never crosses chats");
});

test("compact reset is zero; missing windows do not borrow another model's denominator", async () => {
  const f = runtime();
  await f.emit({ kind: "window", model: "first", contextWindow: 100 });
  await f.emit({ kind: "sample", model: "first", totalTokens: 75 });
  await f.emit({ kind: "reset" });
  assert.equal(f.context()?.totalTokens, 0);
  assert.equal(f.context()?.contextWindow, 100);
  await f.emit({ kind: "sample", model: "second", totalTokens: 12 });
  assert.equal(f.context()?.totalTokens, 12);
  assert.equal(f.context()?.contextWindow, undefined);
});

test("malformed, metadata-only and zero Claude samples preserve known context", async () => {
  const f = runtime();
  await f.emit({ kind: "sample", model: "first", inputTokens: 70, contextWindow: 100 });
  for (const sample of [
    { kind: "sample", model: "first" }, { kind: "sample", model: "first", inputTokens: 0, outputTokens: 0 },
    { kind: "sample", totalTokens: -1 }, { kind: "sample", totalTokens: "80" },
    { kind: "sample", totalTokens: Number.POSITIVE_INFINITY }, { kind: "sample", totalTokens: 2.5 },
    { kind: "window", model: "first", contextWindow: 0 }, { kind: "unknown", totalTokens: 80 },
  ]) await f.emit(sample);
  assert.equal(f.context()?.totalTokens, 70);
  assert.equal(f.context()?.contextWindow, 100);
});

function composer(always: boolean, metadata: boolean, agentType: string, contextUsage?: Sample) {
  const exportsCache = new Map<string, unknown>();
  function load(file: string): unknown {
    const filename = resolve(file);
    if (exportsCache.has(filename)) return exportsCache.get(filename);
    const exports = {};
    exportsCache.set(filename, exports);
    const js = ts.transpileModule(readFileSync(filename, "utf8"), { compilerOptions: { module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2022, jsx: ts.JsxEmit.ReactJSX } }).outputText;
    runInNewContext(js, { exports, console, setTimeout, clearTimeout, require(name: string) {
      if (name === "react" || name === "react/jsx-runtime") return dependencies(name);
      if (name.endsWith(".css")) return {};
      if (name.endsWith("/Tooltip.tsx")) return { Tooltip: ({ children }: { children: unknown }) => children, ShortcutHint: () => null };
      if (name.endsWith("/usePresence.ts")) return { usePresence: (open: boolean) => ({ mounted: open, attrs: {} }) };
      if (name.endsWith("/KeybindingsProvider.tsx")) return { useKeybindingOverrides: () => ({ overrides: {} }), useKeybindings() {} };
      if (name.endsWith("/overrides.ts")) return { effectiveShortcuts };
      if (name.endsWith("/shortcut.ts")) return shortcuts;
      if (name === "./queue.ts") return queue;
      if (name === "./queueKeys.ts") return queueKeys;
      if (name === "./slashCommands.ts") return commands;
      if (name === "./fileMentions.ts") return mentions;
      if (name === "./composerDocument.ts") return document;
      if (name === "./userContent.ts") return userContent;
      if (name === "./ComposerEditor.tsx") return { ComposerEditor: () => null };
      if (name === "./SlashMenu.tsx") return { SlashMenu: () => null };
      if (name === "./FileMentionMenu.tsx") return { FileMentionMenu: () => null };
      if (name === "./composerLongPaste.ts") return { longPasteAttachment: () => null };
      if (name === "./composerPasteFilePaths.ts") return { readClipboardTextFromPlatform: async () => null };
      if (name.endsWith("/useBoolPref.ts")) return { useBoolPref: (key: string) => key === "always_show_context_wheel" ? always : key === "internal_request_metadata" ? metadata : false };
      if (name.endsWith("/useSettings.ts")) return { useSettings: () => ({ settings: { general: {}, experimental: {} } as Settings, save: async () => true }) };
      if (name.endsWith("/prefs.ts")) return { withBoolPref };
      if (name === "../../backend") return { fuzzyFilterFiles: async () => [] };
      if (name === "./composerIcons.tsx") return Object.fromEntries(["ArrowUpIcon", "CircleDotIcon", "CornerDownLeftIcon", "PaperclipIcon", "PlusIcon", "StopCircleIcon"].map((k) => [k, () => null]));
      if (name === "./contextUsage.ts" || name === "./ContextUsage.tsx") return load(resolve(dirname(filename), name));
      throw new Error(`Unexpected context fixture dependency ${name}`);
    } });
    return exports;
  }
  const { Composer } = load("src/features/chat/Composer.tsx") as { Composer: unknown };
  let view: ReturnType<typeof create>;
  act(() => { view = create(React.createElement(Composer, { draft: "", setDraft() {}, inputRef: { current: null }, queue: queue.emptyQueue(), canSteer: false, onSubmit() {}, onStop() {}, onQueue() {}, agentType, contextUsage })); });
  return { button: () => view.root.findAll((n: { type: unknown; props: Record<string, unknown> }) => n.type === "button" && n.props["aria-label"] === "Context usage"), close() { act(() => view.unmount()); } };
}

for (const [label, always, metadata, agent, sample, visible] of [
  ["default missing", false, false, "claude", undefined, false],
  ["off below threshold", false, false, "claude", { totalTokens: 6994, contextWindow: 10000 }, false],
  ["rounded 69.95 reaches threshold", false, false, "claude", { totalTokens: 6995, contextWindow: 10000 }, true],
  ["exact threshold", false, false, "claude", { totalTokens: 70, contextWindow: 100 }, true],
  ["on below threshold", true, false, "claude", { totalTokens: 2, contextWindow: 100 }, true],
  ["on missing is unknown", true, false, "claude", undefined, true],
  ["metadata gate", false, true, "claude", undefined, true],
  ["ACP gate", false, false, "acp", undefined, true],
  ["unknown denominator is not a percentage", false, false, "codex", { totalTokens: 95 }, false],
] as const) {
  test(`actual Composer context display: ${label}`, () => {
    const f = composer(always, metadata, agent, sample);
    try { assert.equal(f.button().length, visible ? 1 : 0, "the real Composer must apply the retained context-display gate"); }
    finally { f.close(); }
  });
}
