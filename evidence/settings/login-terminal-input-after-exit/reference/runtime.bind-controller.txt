bindController(e,s){e.onChange(n=>{n.connectionState==="disposed"?(this.sessions.delete(n.id),this.startedAt.delete(n.id),this.clearScheduledPersistence(n.id)):this.schedulePersistence(e,s),n.connectionState==="exited"&&this.flushScheduledPersistence(n.id,{flushToStorage:!0,markEnded:!0}).fireAndForget(),this.sessionChangeEmitter.emit(n)}),e.onData(n=>{this.sessionDataEmitter.emit(n),this.schedulePersistence(e,s)})}
