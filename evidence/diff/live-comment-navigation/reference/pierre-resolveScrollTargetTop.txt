resolveScrollTargetTop(e){if(e.type==="position"){const r=this.clampScrollTop(e.position);return r!==e.position?r:this.clampScrollTop(e.position-this.getStickyHeaderOffset())}const n=this.idToItem.get(e.id);if(n==null){console.warn(`CodeView.scrollTo: unknown item id "${e.id}"`);return}if(e.type==="item")return this.clampScrollTop(this.resolveAlignedScrollPosition(n.top,n.height,e.align,e.offset));if(e.type==="range"){const r=this.getRangeScrollPosition(n,e);if(r==null){console.warn(`CodeView.scrollTo: unable to resolve range ${oDn(e.range)} for item "${e.id}"`);return}return this.clampScrollTop(this.resolveAlignedScrollPosition(n.top+r.top,r.height,e.align,e.offset,this.getStickyHeaderOffset()))}const i=this.getLineScrollPosition(n,e);if(i==null){console.warn(`CodeView.scrollTo: unable to resolve line ${e.lineNumber} for item "${e.id}"`);return}return this.clampScrollTop(this.resolveAlignedScrollPosition(n.top+i.top,i.height,e.align,e.offset,this.getStickyHeaderOffset()))}
