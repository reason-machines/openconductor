type TextNode = string | { children: readonly TextNode[] } | TextNode[];

function evaluate(source: string, filename: string, require: (name: string) => unknown, globals: Record<string, unknown> = {}) {
  const javascript = ts.transpileModule(source, { fileName: filename, compilerOptions: {
    jsx: ts.JsxEmit.ReactJSX, module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2022, esModuleInterop: true,
  } }).outputText;
  const module: { exports: Record<string, unknown> } = { exports: {} };
  runInNewContext(`(function(require, module, exports) {\n${javascript}\n})`, {
    console, Map, Set, Error, crypto: { randomUUID },
    setInterval: unexpected, clearInterval: unexpected, ...globals,
  }, { filename })(require, module, module.exports);
  return module.exports;
}

for (const mode of ["flat", "tree"] satisfies ("flat" | "tree")[]) {
  test(`${mode} rows count active local and remote comments, not sent, resolved, outdated or PR-level rows`, async (t) => {
    const f = await fixture([
      comment("local"), comment("remote", { remote_url: "https://github.com/example/repo/pull/1#discussion_r2", author: "reviewer" }),
      comment("sent", { state: "sent" }), comment("resolved", { is_resolved: true }),
      comment("outdated", { is_outdated: true }), comment("pr", { file_path: "", file_level: true }),
    ], { mode }); t.after(f.close);
    assert.deepEqual(f.counts(paths[0]), ["2"]);
    assert.deepEqual(f.counts(paths[1]), []);
  });
  test(`${mode} zero-comment rows keep their ordinary diff counts and no comment badge`, async (t) => {
    const f = await fixture([], { mode }); t.after(f.close);
    assert.deepEqual(f.counts(paths[0]), []);
    assert.ok(text(f.row(paths[0])).includes("+1"));
    assert.equal(f.row(paths[0]).findByProps({ "aria-label": "modified" }).type, "svg");
  });
}

test("counts stay keyed by exact path when basenames match and replies add another comment", async (t) => {
  const f = await fixture([comment("root"), comment("reply", { thread_id: "root", reply_to_comment_id: "root" }), comment("other", { file_path: paths[1] })]); t.after(f.close);
  assert.deepEqual(f.counts(paths[0]), ["2"]);
  assert.deepEqual(f.counts(paths[1]), ["1"]);
});

test("the count predicate does not invent a nonempty-body or visible-diff-line requirement", async (t) => {
  const f = await fixture([comment("empty", { body: "", line_number: 999 })]); t.after(f.close);
  assert.deepEqual(f.counts(paths[0]), ["1"]);
});

test("saving through the actual new-draft composer updates the row count and normal payload", async (t) => {
  const f = await fixture([]); t.after(f.close);
  await f.saveDraft("new review comment");
  const saved = await f.saved();
  assert.equal(saved.length, 1);
  assert.equal(saved[0].body, "new review comment");
  assert.equal(saved[0].file_path, paths[0]);
  assert.deepEqual(f.counts(paths[0]), ["1"]);
});

test("adding comments to chat preserves the real attachment and leaves no pending count", async (t) => {
  const f = await fixture([comment("whole", { file_level: true })]); t.after(f.close);
  await f.addAll();
  assert.equal(f.attached.length, 1);
  assert.equal((await f.saved())[0].state, "sent");
  assert.deepEqual(f.counts(paths[0]), []);
});

test("switching workspaces replaces the counts for identical relative paths", async (t) => {
  const f = await fixture([comment("one"), comment("two-a", { workspace_id: "two" }), comment("two-b", { workspace_id: "two" })]); t.after(f.close);
  assert.deepEqual(f.counts(paths[0]), ["1"]);
  await f.update({ workspace: "two" });
  assert.deepEqual(f.counts(paths[0]), ["2"]);
  await f.update({ workspace: "one" });
  assert.deepEqual(f.counts(paths[0]), ["1"]);
});

test("filtering and changing list modes preserve each visible file's count", async (t) => {
  const f = await fixture([comment("a"), comment("b", { file_path: paths[1] })]); t.after(f.close);
  await f.update({ query: "test/example.ts", mode: "tree" });
  assert.deepEqual(f.counts(paths[1]), ["1"]);
  await f.update({ query: "", mode: "flat" });
  assert.deepEqual(f.counts(paths[0]), ["1"]);
  assert.deepEqual(f.counts(paths[1]), ["1"]);
});

test("ordinary file selection still reaches the existing navigation action", async (t) => {
  const f = await fixture([]); t.after(f.close);
  await act(async () => f.row(paths[1]).props.onClick());
  assert.deepEqual(f.opened, [paths[1]]);
});

test("the existing multi-file summary remains independent of comment counts", async (t) => {
  const f = await fixture([]); t.after(f.close);
  assert.ok(f.allText().includes("3 files changed"));
  assert.ok(f.allText().includes("+3"));
});
