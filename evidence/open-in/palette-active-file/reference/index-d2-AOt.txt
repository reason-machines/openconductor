function AOt(t){if(t.activeTabType==="special"){if(t.specialTabMode==="file"&&t.specialTabPath)return{type:"file",path:t.specialTabPath,mode:t.specialTabFileMode??"diff",...t.specialTabLineNumber!==void 0&&{lineNumber:t.specialTabLineNumber}};if(t.specialTabMode==="comment"&&t.specialTabCommentId)return{type:"comment",commentId:t.specialTabCommentId};if(t.specialTabMode==="browser"&&t.specialTabBrowserUrl)return{type:"browser",url:t.specialTabBrowserUrl};if(t.specialTabMode==="desktop")return{type:"desktop"};if(t.specialTabMode==="pr")return{type:"pr"}}return t.activeTabType==="terminal"?{type:"terminal",terminalTabId:t.terminalTabId}:t.activeTabType==="notes"?{type:"notes"}:{type:"session",sessionId:t.sessionId}}
