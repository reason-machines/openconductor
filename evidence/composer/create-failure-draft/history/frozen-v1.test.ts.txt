function deferred<T>() {
  let resolve!: (value: T) => void;
  let reject!: (reason: unknown) => void;
  const promise = new Promise<T>((yes, no) => { resolve = yes; reject = no; });
  return { promise, resolve, reject };
}

const original = "  Keep this exact prompt\nincluding café and trailing spaces  ";

test("a rejected dismissed modal exposes exact captured prompt through the real toast action", async () => {
  const view = await fixture();
  try { view.message(original); await view.submit(); assert.equal(view.open, false); await view.fail(); assert.equal(view.pending.size, 0); assert.equal(view.warnings.length, 1); await view.recover(); assert.deepEqual(view.clipboard, [original]); } finally { view.dispose(); }
});

test("recovering a detached failure never replaces the new composer's draft or selected target", async () => {
  const view = await fixture();
  try { view.message(original); await view.submit(); await view.replacement("New unrelated draft"); await view.fail(); assert.equal(view.open, true); assert.equal(view.draft, "New unrelated draft"); assert.equal(view.repo, "other-repo"); assert.equal(view.target, "release"); await view.recover(); assert.deepEqual(view.clipboard, [original]); assert.equal(view.draft, "New unrelated draft"); assert.equal(view.repo, "other-repo"); } finally { view.dispose(); }
});

test("full-page rejection retains an editable draft and the same repository and target", async () => {
  const view = await fixture({ page: true });
  try { view.message(original); await view.submit(); await view.fail(); assert.equal(view.open, true); assert.equal(view.draft, original); assert.equal(view.repo, "repo"); assert.equal(view.target, "main"); assert.deepEqual(view.warnings, []); view.message("Retry draft"); await view.submit(); assert.equal(view.requests.length, 2); } finally { view.dispose(); }
});

test("pending full-page creation suppresses duplicate submit and keeps the captured request", async () => {
  const view = await fixture({ page: true });
  try { view.message(original); await view.submit(); await view.submit(); assert.equal(view.requests.length, 1); assert.equal(view.pending.size, 1); assert.deepEqual(view.warnings, []); assert.deepEqual(view.clipboard, []); await view.fail(); assert.equal(view.pending.size, 0); } finally { view.dispose(); }
});

test("empty modal prompt reports the error without a recover action", async () => {
  const view = await fixture();
  try { await view.submit(); await view.fail(); assert.deepEqual(view.warnings, ["Failed to create workspace: Synthetic create rejected"]); assert.equal(view.recoverButtons().length, 0); assert.deepEqual(view.clipboard, []); assert.equal(view.pending.size, 0); } finally { view.dispose(); }
});

test("string rejection uses the actual errorText formatter and leaves no pending row", async () => {
  const view = await fixture();
  try { await view.submit(); await view.fail(0, "branch exists"); assert.deepEqual(view.warnings, ["Failed to create workspace: branch exists"]); assert.equal(view.pending.size, 0); assert.equal(view.recoverButtons().length, 0); } finally { view.dispose(); }
});

test("successful modal creation produces no failure recovery or clipboard write", async () => {
  const view = await fixture();
  try { view.message(original); await view.submit(); await view.succeed(); assert.equal(view.open, false); assert.deepEqual(view.warnings, []); assert.equal(view.recoverButtons().length, 0); assert.deepEqual(view.clipboard, []); assert.equal(view.pending.size, 0); } finally { view.dispose(); }
});

test("no selected repository never launches or emits a recovery error", async () => {
  const view = await fixture({ noRepository: true });
  try { view.message(original); await view.submit(); assert.equal(view.open, true); assert.equal(view.requests.length, 0); assert.equal(view.pending.size, 0); assert.deepEqual(view.warnings, []); assert.deepEqual(view.clipboard, []); } finally { view.dispose(); }
});
