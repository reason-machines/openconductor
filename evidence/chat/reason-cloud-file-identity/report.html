const toolDisclosure = (() => {
  let expanded: ReadonlyMap<string, boolean> = new Map();
  const listeners = new Set<() => void>();
  return {
    getSnapshot: () => expanded,
    subscribe(listener: () => void) {
      listeners.add(listener);
      return () => { listeners.delete(listener); };
    },
    toggle(id: string, defaultOpen: boolean) {
      expanded = new Map(expanded).set(id, !isToolExpanded(id, expanded, defaultOpen));
      for (const listener of listeners) listener();
    },
  };
})();

function groupTurns(rows: DisplayRow[]): Turn[] {
  const out: Turn[] = [];
  for (const row of rows) {
    const turnId = rowTurn(row);
    const last = out[out.length - 1];
    if (last && last.turnId === turnId && !last.rows.some((r) => r.kind === "turn_end")) last.rows.push(row);
    else out.push({ key: `turn:${row.key}`, turnId, rows: [row] });
  }
  return out;
}
