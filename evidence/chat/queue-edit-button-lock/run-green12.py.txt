"""Identical original pointer12 fixture, candidate Composer only; separate admission required."""
from pathlib import Path
import argparse
import datetime
import hashlib
import json
import os
import resource
import shutil
import signal
import subprocess

D = Path('.qa/queue-edit-button-lock-135/proposal-worktree/pointer-review')
p = argparse.ArgumentParser()
p.add_argument('--reservation', required=True)
a = p.parse_args()
sha = lambda data: hashlib.sha256(data).hexdigest()
raw = (D / 'green12-binding.json').read_bytes()
r = json.loads(raw)
assert r['phase'] == 'IDENTICAL_ORIGINAL_GREEN12_CANDIDATE_ONLY'
assert sha(Path(__file__).read_bytes()) == r['supervisor_sha256']
now = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()
out = D / 'runtime-green12'
out.mkdir(exist_ok=False)
receipt = {'reservation': a.reservation, 'started_utc': now(), 'request_sha256': sha(raw),
           'command': r['command'], 'cwd': r['cwd'], 'node_test_executed': False}
proc = None
pins = {pin['path']: pin for pin in r['pins']}

def verify():
    assert len(pins) == r['unique_pin_count']
    for pin in pins.values():
        data = Path(pin['path']).read_bytes()
        assert len(data) == pin['bytes'] and sha(data) == pin['sha256'], pin['path']
    candidate = Path(r['candidate_source']).read_text()
    for item in r['candidate_inverse_replacements']:
        assert candidate.count(item['from']) == 1, 'inverse source anchor not unique'
        candidate = candidate.replace(item['from'], item['to'], 1)
    assert candidate.encode() == Path(r['baseline_source']).read_bytes(), 'candidate is not exactly the approved minimal diff'
    assert git('-C', r['cwd'], 'rev-parse', 'HEAD').decode().strip() == r['candidate_commit'], 'candidate commit drift'
    assert git('-C', r['cwd'], 'show', r['candidate_commit'] + ':src/features/chat/Composer.tsx') == Path(r['candidate_source']).read_bytes()
    assert git('-C', r['cwd'], 'show', r['candidate_commit'] + ':src/features/chat/composerQueueEditButtonLock.test.ts') == Path(r['permanent_port']).read_bytes()
    return len(pins)

def members(group):
    found = []
    for path in Path('/proc').iterdir():
        if not path.name.isdigit():
            continue
        try:
            fields = (path / 'stat').read_text().rsplit(')', 1)[1].split()
            if int(fields[2]) == group and fields[0] != 'Z':
                found.append(int(path.name))
        except (OSError, ValueError):
            pass
    return found

def git(*args):
    return subprocess.check_output(['git', '-c', 'gc.auto=0', *args])

try:
    assert 0 in os.sched_getaffinity(0), 'CPU0 unavailable'
    receipt['free_bytes_before'] = shutil.disk_usage('.').free
    assert receipt['free_bytes_before'] >= 4 * 1024 ** 3, 'free disk below 4GiB'
    own = []
    for path in Path('/proc').iterdir():
        if not path.name.isdigit():
            continue
        try:
            if ((path / 'cwd').resolve() in (Path.cwd(), Path(r['cwd']).resolve())
                    and (path / 'comm').read_text().strip() in ['node', 'npm', 'cargo', 'rustc', 'bun', 'vite']):
                own.append(int(path.name))
        except OSError:
            pass
    receipt['own_runtime_jobs_before'] = own
    assert not own, 'own runtime already present'
    receipt['preflight_pin_checks'] = verify()
    main = git('ls-remote', 'origin', 'refs/heads/main').decode().split()[0]
    receipt['actual_main'] = main
    assert git('rev-parse', 'origin/main').decode().strip() == main, 'fresh main not fetched'
    receipt['source_equivalent_checks'] = 0
    for path in r['fresh_main_source_paths']:
        assert git('show', main + ':' + path) == Path(path).read_bytes(), path + ' differs from actual main'
        receipt['source_equivalent_checks'] += 1

    def limits():
        resource.setrlimit(resource.RLIMIT_FSIZE, (8 * 1024 ** 2, 8 * 1024 ** 2))

    with (out / 'test.stdout').open('wb') as stdout, (out / 'test.stderr').open('wb') as stderr:
        receipt['test_started_utc'] = now()
        proc = subprocess.Popen(r['command'], stdout=stdout, stderr=stderr, cwd=r['cwd'],
                                start_new_session=True, preexec_fn=limits)
        receipt['process_group'] = proc.pid
        receipt['node_test_executed'] = True
        try:
            receipt['exit_code'] = proc.wait(timeout=67)
        except subprocess.TimeoutExpired:
            receipt['wrapper_timeout'] = True
            os.killpg(proc.pid, signal.SIGTERM)
            try:
                proc.wait(timeout=5)
            except subprocess.TimeoutExpired:
                os.killpg(proc.pid, signal.SIGKILL)
                proc.wait(timeout=5)
            receipt['exit_code'] = proc.returncode
        receipt['test_terminated_utc'] = now()
except Exception as error:
    receipt['setup_or_wrapper_error'] = str(error)
finally:
    if proc is not None and members(proc.pid):
        os.killpg(proc.pid, signal.SIGKILL)
        if proc.poll() is None:
            proc.wait(timeout=5)
    receipt['surviving_pids'] = [] if proc is None else members(proc.pid)
    receipt['actual_cleanup_release_utc'] = now()
    receipt['child_maxrss_kib_not_aggregate'] = resource.getrusage(resource.RUSAGE_CHILDREN).ru_maxrss
    receipt['rss_scope'] = 'Child maximum RSS, not aggregate process-group RSS or a group cap; heap768MiB per Node process.'
    try:
        receipt['postrun_pin_checks'] = verify()
    except Exception as error:
        receipt['postrun_drift_error'] = str(error)
    receipt['outputs'] = []
    for path in sorted(out.glob('test.*')):
        data = path.read_bytes()
        receipt['outputs'].append({'path': str(path), 'bytes': len(data), 'sha256': sha(data)})
    receipt['retained_output_bytes'] = sum(item['bytes'] for item in receipt['outputs'])
    receipt['within_output_16mib'] = receipt['retained_output_bytes'] <= 16 * 1024 ** 2
    (out / 'receipt.json').write_text(json.dumps(receipt, indent=2) + '\n')
    print(json.dumps(receipt, indent=2))
